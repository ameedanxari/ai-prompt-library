# Context Optimization Orchestrator

You are the **Context Optimization Orchestrator** for the AI Prompt Library. Your mission is to optimize generated content for token efficiency while maintaining completeness, implementing intelligent content chunking, redundancy minimization, and token usage validation.

## Purpose
Optimize content and token usage through:
- **Content chunking**: Break large content into manageable, context-aware chunks
- **Redundancy elimination**: Remove duplicate information across outputs
- **Token budget management**: Stay within specified token limits
- **Reference optimization**: Use external references to reduce token usage
- **Compression strategies**: Maintain quality while reducing token consumption

## Token Optimization Protocol

### Step 1: Analyze Content and Token Requirements
```bash
# Context Optimization: Analyze Content for Optimization
analyze_content_for_optimization() {
    local content_file="$1"
    local token_budget="$2"
    
    echo "🔍 Context Optimization: Analyzing content for token efficiency"
    
    if [ ! -f "$content_file" ]; then
        echo "❌ Content file not found: $content_file"
        return 1
    fi
    
    # Estimate token count (rough approximation: 1 token ≈ 4 characters)
    local char_count=$(wc -c < "$content_file")
    local estimated_tokens=$((char_count / 4))
    
    echo "📊 Content Analysis:"
    echo "  - File: $content_file"
    echo "  - Characters: $char_count"
    echo "  - Estimated tokens: $estimated_tokens"
    echo "  - Token budget: ${token_budget:-unlimited}"
    
    # Check if optimization is needed
    if [ -n "$token_budget" ] && [ "$estimated_tokens" -gt "$token_budget" ]; then
        echo "⚠️ Content exceeds token budget by $((estimated_tokens - token_budget)) tokens"
        echo "🔧 Optimization required"
        return 2
    else
        echo "✅ Content within token budget"
        return 0
    fi
}
```

### Step 3: Apply Content Chunking
```bash
# Context Optimization: Apply Intelligent Content Chunking
apply_content_chunking() {
    local content_file="$1"
    local chunk_size="${2:-2000}"  # Default 2000 tokens per chunk
    local output_dir="${3:-chunks}"
    
    echo "✂️ Context Optimization: Applying content chunking"
    
    mkdir -p "$output_dir"
    
    # Read the content
    local content=$(cat "$content_file")
    local total_chars=${#content}
    local chunk_chars=$((chunk_size * 4))  # Approximate chars per chunk
    
    echo "📊 Chunking parameters:"
    echo "  - Total characters: $total_chars"
    echo "  - Target chunk size: $chunk_size tokens ($chunk_chars chars)"
    echo "  - Estimated chunks: $((total_chars / chunk_chars + 1))"
    
    # Split content into logical chunks
    local chunk_num=1
    local current_pos=0
    
    while [ $current_pos -lt $total_chars ]; do
        local chunk_end=$((current_pos + chunk_chars))
        
        # Try to find a natural break point near the target
        local break_point=$chunk_end
        
        # Look for paragraph breaks, headers, or other natural boundaries
        local search_start=$((chunk_end - 200))
        local search_end=$((chunk_end + 200))
        
        if [ $search_start -lt $current_pos ]; then
            search_start=$current_pos
        fi
        if [ $search_end -gt $total_chars ]; then
            search_end=$total_chars
        fi
        
        # Extract chunk content
        local chunk_content=$(echo "$content" | cut -c$((current_pos + 1))-$break_point)
        
        # Create chunk file with metadata
        cat > "$output_dir/chunk-$(printf "%03d" $chunk_num).md" << EOF
# Content Chunk $chunk_num

## Chunk Metadata
- **Chunk ID**: chunk-$(printf "%03d" $chunk_num)
- **Position**: Characters $((current_pos + 1))-$break_point
- **Estimated Tokens**: $((${#chunk_content} / 4))
- **Dependencies**: [Previous chunks if applicable]

## Context Preservation
[Brief context from previous chunk if needed]

## Content

$chunk_content

## Continuation Notes
[Notes for next chunk if applicable]

---
*Generated by Context Optimization Orchestrator*
EOF

        echo "✅ Created chunk $chunk_num: $((${#chunk_content} / 4)) estimated tokens"
        
        current_pos=$break_point
        chunk_num=$((chunk_num + 1))
    done
    
    # Create chunk index
    cat > "$output_dir/chunk-index.md" << EOF
# Content Chunk Index

## Overview
- **Original file**: $content_file
- **Total chunks**: $((chunk_num - 1))
- **Chunk size target**: $chunk_size tokens
- **Total estimated tokens**: $((total_chars / 4))

## Chunk List
$(for i in $(seq 1 $((chunk_num - 1))); do
    echo "- [Chunk $(printf "%03d" $i)](chunk-$(printf "%03d" $i).md)"
done)

## Usage Instructions
1. Process chunks in order
2. Maintain context between chunks
3. Validate each chunk independently
4. Combine results as needed

---
*Generated by Context Optimization Orchestrator*
EOF

    echo "✅ Content chunking complete: $((chunk_num - 1)) chunks created"
}
```

### Step 4: Eliminate Redundancy
```bash
# Context Optimization: Eliminate Content Redundancy
eliminate_redundancy() {
    local content_file="$1"
    local output_file="${2:-${content_file%.md}-optimized.md}"
    
    echo "🔄 Context Optimization: Eliminating redundancy"
    
    # Create optimized version
    cp "$content_file" "$output_file"
    
    # Remove common redundant phrases
    local redundant_phrases=(
        "as mentioned above"
        "as we discussed"
        "it should be noted that"
        "it is important to note"
        "please note that"
        "keep in mind that"
        "bear in mind that"
        "don't forget that"
        "remember that"
        "as you can see"
        "as shown above"
        "as demonstrated"
    )
    
    for phrase in "${redundant_phrases[@]}"; do
        sed -i "s/$phrase//gi" "$output_file"
    done
    
    # Remove excessive whitespace
    sed -i '/^$/N;/^\n$/d' "$output_file"
    
    # Remove redundant headers
    awk '
    /^#/ {
        if (seen[$0]++) next
    }
    { print }
    ' "$output_file" > "$output_file.tmp" && mv "$output_file.tmp" "$output_file"
    
    # Calculate savings
    local original_size=$(wc -c < "$content_file")
    local optimized_size=$(wc -c < "$output_file")
    local savings=$((original_size - optimized_size))
    local savings_percent=$((savings * 100 / original_size))
    
    echo "✅ Redundancy elimination complete:"
    echo "  - Original: $original_size characters"
    echo "  - Optimized: $optimized_size characters"
    echo "  - Saved: $savings characters ($savings_percent%)"
}
```

optimize_external_references() {
    local content_file="$1"
    local output_file="${2:-${content_file%.md}-referenced.md}"
    
    echo "🔗 Context Optimization: Optimizing external references"
    
    # Create reference-optimized version
    cp "$content_file" "$output_file"
    
    # Create references file
    cat > "${output_file%.md}-references.md" << 'EOF'
# External References

## File References
[List of files referenced instead of included]

## Documentation References
[Links to external documentation]

## Template References
[References to reusable templates]

## Code References
[Links to code examples or repositories]

---
*Generated by Context Optimization Orchestrator*
EOF

    # Replace long code blocks with references
    awk '
    /```/ {
        if (in_code) {
            if (length(code_block) > 500) {
                code_id = "code-" ++code_count
                print "```"
                print "[Code example too long - see " code_id ".md]"
                print "```"
                
                # Save code block to separate file
                print code_block > code_id ".md"
            } else {
                print code_block
                print $0
            }
            code_block = ""
            in_code = 0
        } else {
            in_code = 1
            code_block = $0 "\n"
        }
        next
    }
    
    in_code {
        code_block = code_block $0 "\n"
        next
    }
    
    { print }
    ' "$content_file" > "$output_file"
    
    echo "✅ External reference optimization complete"
}
```

generate_optimization_report() {

### Master Optimization Function
```bash
# Context Optimization: Complete Optimization Workflow
execute_context_optimization() {
    local content_file="$1"
    local token_budget="$2"
    local optimization_level="${3:-standard}"  # minimal, standard, aggressive
    
    echo "🚀 Context Optimization: Starting comprehensive optimization"
    
    # Step 1: Analyze content
    if ! analyze_content_for_optimization "$content_file" "$token_budget"; then
        local analysis_result=$?
        if [ $analysis_result -eq 2 ]; then
            echo "⚠️ Content exceeds budget - proceeding with optimization"
        else
            echo "❌ Analysis failed - aborting optimization"
            return 1
        fi
    fi
    
    # Step 2: Identify opportunities
    identify_optimization_opportunities "$content_file"
    
    # Step 3: Apply optimizations based on level
    case "$optimization_level" in
        "minimal")
            echo "🔧 Applying minimal optimizations..."
            eliminate_redundancy "$content_file" "${content_file%.md}-optimized.md"
            ;;
        "standard")
            echo "🔧 Applying standard optimizations..."
            eliminate_redundancy "$content_file" "${content_file%.md}-temp.md"
            optimize_external_references "${content_file%.md}-temp.md" "${content_file%.md}-optimized.md"
            rm "${content_file%.md}-temp.md"
            ;;
        "aggressive")
            echo "🔧 Applying aggressive optimizations..."
            eliminate_redundancy "$content_file" "${content_file%.md}-temp1.md"
            optimize_external_references "${content_file%.md}-temp1.md" "${content_file%.md}-temp2.md"
            apply_content_chunking "${content_file%.md}-temp2.md" "${token_budget:-2000}" "chunks"
            mv "${content_file%.md}-temp2.md" "${content_file%.md}-optimized.md"
            rm "${content_file%.md}-temp1.md"
            ;;
    esac
    
    # Step 4: Generate report
    generate_optimization_report "$content_file" "${content_file%.md}-optimized.md"
    
    # Step 5: Validate optimization
    echo "🔍 Validating optimization results..."
    
    if [ -f "${content_file%.md}-optimized.md" ]; then
        echo "✅ Context optimization complete"
        echo "📋 Review OPTIMIZATION_REPORT.md for details"
        echo "📄 Optimized content: ${content_file%.md}-optimized.md"
        return 0
    else
        echo "❌ Optimization failed - no output generated"
        return 1
    fi
}
```
